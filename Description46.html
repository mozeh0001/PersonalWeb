Machine Learning Rocket Detection


As part of RPL Avionics, I am developing a YOLO11 object detection system for identifying the team's rocket in images and launch footage.


  • Dataset & Annotation: Built a custom rocket dataset and created YOLO bounding-box annotations.
  • Preprocessing: Developed Python tools to convert HEIC, HEIF, and Sony ARW images for training.
  • Model Training: Fine-tuned YOLO11 on the custom dataset and evaluated performance on validation images.
  • Detection: Built image and video inference tools that locate rockets and visualize confidence-scored predictions.

Model Training 


from ultralytics import YOLO




# we start with a pretrained YOLO model
# it already understands many general visual features
#
# we will retrain it specifically to detect rockets
model = YOLO("yolo11n.pt")



# train model
results = model.train(

    # tells YOLO where our dataset is
    data="data.yaml",

    # number of times YOLO sees the complete training dataset
    epochs=50,

    # resize images to 640 x 640 while training
    imgsz=640,

    # number of images processed together
    batch=16,

    # name of this training experiment
    name="rocket_detector_v5"
)


print("Rocket detector training complete!")


Image Preprocessing & Conversion 


from pathlib import Path

from PIL import Image
from pillow_heif import register_heif_opener
import rawpy


# allow Pillow to open HEIC / HEIF
register_heif_opener()


# folders to check
IMAGE_FOLDERS = [
    Path("dataset/images/train"),
    Path("dataset/images/val")
]


# convert images
for folder in IMAGE_FOLDERS:

    print()
    print(f"Checking folder: {folder}")

    if not folder.exists():
        print(f"Folder does not exist: {folder}")
        continue

    for image_path in folder.iterdir():

        extension = image_path.suffix.lower()

        # HEIC / HEIF -> JPG
        if extension in [".heic", ".heif"]:

            print(f"Found HEIC: {image_path.name}")

            try:
                image = Image.open(image_path)
                image = image.convert("RGB")

                output_path = image_path.with_suffix(".jpg")

                image.save(
                    output_path,
                    "JPEG",
                    quality=95
                )

                print(f"Converted -> {output_path.name}")

            except Exception as error:
                print(f"FAILED: {image_path.name}")
                print(error)

        # ARW -> JPG
        elif extension == ".arw":

            print(f"Found ARW: {image_path.name}")

            try:
                with rawpy.imread(str(image_path)) as raw:

                    rgb = raw.postprocess()

                    image = Image.fromarray(rgb)

                    output_path = image_path.with_suffix(".jpg")

                    image.save(
                        output_path,
                        "JPEG",
                        quality=95
                    )

                print(f"Converted -> {output_path.name}")

            except Exception as error:
                print(f"FAILED: {image_path.name}")
                print(error)


print()
print("Finished converting images!")


Video Detection & Tracking 


from ultralytics import YOLO
from tkinter import Tk
from tkinter.filedialog import askopenfilename
import cv2



# load trained model
model = YOLO(
    "runs/detect/rocket_detector_v5/weights/best.pt"
)



# select video
root = Tk()
root.withdraw()

video_path = askopenfilename(
    title="Choose a rocket video",
    filetypes=[
        ("Video Files", "*.mp4 *.avi *.mov *.mkv")
    ]
)

root.destroy()

if not video_path:
    print("No video selected.")
    quit()



# open video
cap = cv2.VideoCapture(video_path)

if not cap.isOpened():
    print("Could not open video.")
    quit()


# get original video information
fps = cap.get(cv2.CAP_PROP_FPS)

width = int(
    cap.get(cv2.CAP_PROP_FRAME_WIDTH)
)

height = int(
    cap.get(cv2.CAP_PROP_FRAME_HEIGHT)
)

print()
print(f"Video resolution: {width} x {height}")
print(f"Video FPS: {fps:.2f}")



# create output video
output_path = "rocket_detection_output.mp4"

fourcc = cv2.VideoWriter_fourcc(
    *"mp4v"
)

writer = cv2.VideoWriter(
    output_path,
    fourcc,
    fps,
    (width, height)
)



# display settings
MAX_DISPLAY_WIDTH = 1920
MAX_DISPLAY_HEIGHT = 1080



# process video frame by frame
frame_number = 0


while True:

    success, frame = cap.read()

    if not success:
        break

    frame_number += 1


    # running yolo
    results = model.predict(
        source=frame,

        # inimum confidence
        conf=0.15,

        # dont print YOLO information for every frame
        verbose=False
    )


    boxes = results[0].boxes


    # keep only the best rocket detection
    if len(boxes) > 0:

        best_box = max(
            boxes,
            key=lambda box: float(box.conf[0])
        )

        confidence = float(
            best_box.conf[0]
        )

        x1, y1, x2, y2 = best_box.xyxy[0]

        x1 = int(x1)
        y1 = int(y1)
        x2 = int(x2)
        y2 = int(y2)



        # draw bounding box
        cv2.rectangle(
            frame,
            (x1, y1),
            (x2, y2),
            (0, 255, 0),
            3
        )

        # draw confidence text
        cv2.putText(
            frame,
            f"rocket {confidence * 100:.1f}%",
            (x1, max(y1 - 10, 30)),
            cv2.FONT_HERSHEY_SIMPLEX,
            1,
            (0, 255, 0),
            2
        )



    # save full resolution frame
    writer.write(frame)


    # make smaller copy for display
    display_frame = frame.copy()

    scale = min(
        MAX_DISPLAY_WIDTH / width,
        MAX_DISPLAY_HEIGHT / height,
        1.0
    )

    display_width = int(
        width * scale
    )

    display_height = int(
        height * scale
    )

    display_frame = cv2.resize(
        display_frame,
        (display_width, display_height)
    )

    # show video
    cv2.imshow(
        "Rocket Video Detection",
        display_frame
    )


    # press Q to stop the video early
    if cv2.waitKey(1) & 0xFF == ord("q"):
        break



# clean up
cap.release()
writer.release()

cv2.destroyAllWindows()



print()
print("Finished processing video!")
print(f"Frames processed: {frame_number}")
print(f"Saved to: {output_path}")


Custom Bounding Box Annotation Tool 


from pathlib import Path
import cv2



# settings
SPLIT = "val"

IMAGE_DIR = Path(f"dataset/images/{SPLIT}")
LABEL_DIR = Path(f"dataset/labels/{SPLIT}")

LABEL_DIR.mkdir(parents=True, exist_ok=True)

# maximum size of the labeling window
MAX_WIDTH = 1000
MAX_HEIGHT = 700


# get images

image_files = []

for extension in ["*.jpg", "*.jpeg", "*.png", "*.JPG", "*.JPEG", "*.PNG"]:
    image_files.extend(IMAGE_DIR.glob(extension))

image_files = sorted(image_files)

print(f"Found {len(image_files)} images.")



# label each image
for image_path in image_files:

    label_path = LABEL_DIR / f"{image_path.stem}.txt"

    # don't make us label an image again
    if label_path.exists():
        print(f"Already labeled: {image_path.name}")
        continue

    image = cv2.imread(str(image_path))

    if image is None:
        print(f"Could not open: {image_path.name}")
        continue

    # original image dimensions
    original_height, original_width = image.shape[:2]



    # shrink the image for display
    scale = min(
        MAX_WIDTH / original_width,
        MAX_HEIGHT / original_height,
        1.0
    )

    display_width = int(original_width * scale)
    display_height = int(original_height * scale)

    display_image = cv2.resize(
        image,
        (display_width, display_height)
    )



    # draw box
    print()
    print(f"Image: {image_path.name}")
    print("Draw a tight box around the rocket.")
    print("Press ENTER or SPACE when finished.")

    x, y, w, h = cv2.selectROI(
        "Label Rocket",
        display_image,
        showCrosshair=True,
        fromCenter=False
    )

    cv2.destroyAllWindows()

    if w == 0 or h == 0:
        print("No box selected.")
        continue


    # convert back to normal image size
    x = x / scale
    y = y / scale
    w = w / scale
    h = h / scale


    # convert to yolo format
    center_x = (x + w / 2) / original_width
    center_y = (y + h / 2) / original_height

    normalized_width = w / original_width
    normalized_height = h / original_height


    # Save label

    # 0 = rocket
    with open(label_path, "w") as file:
        file.write(
            f"0 "
            f"{center_x:.6f} "
            f"{center_y:.6f} "
            f"{normalized_width:.6f} "
            f"{normalized_height:.6f}\n"
        )

    print(f"Saved: {label_path}")


print()
print("Finished labeling!")

Machine Learning Pipeline


The current pipeline takes raw rocket imagery through dataset preparation, supervised training, validation, and inference.


Rocket Images  →  Bounding Box Labels  →  YOLO11 Fine-Tuning  →  Validation  →  Rocket Detection


During supervised training, the bounding boxes act as the ground-truth examples that tell the model where the rocket is located. After training, the resulting model can predict a rocket's location in new images without being given a bounding box beforehand.



YOLO Dataset & Training Structure


Training and validation images are paired with YOLO bounding-box labels, while each training run generates model weights and evaluation results.



Model Training


The model trains for 50 epochs while tracking loss, precision, recall, and mAP on the validation dataset.

Training & Evaluation


I separate the dataset into training and validation images so that model performance can be evaluated on examples that are not used to directly update the network during training.


  • Training Epochs (x50): The model repeatedly processes the training dataset while updating its learned parameters.
  • Validation: A separate set of labeled images is used to evaluate how well the model generalizes beyond its training examples.
  • Confidence Scores: Each detection includes a confidence value representing the model's confidence that the predicted region contains a rocket.
  • Performance Metrics: Training outputs include precision, recall, F1, and mean Average Precision metrics for evaluating detection performance.
  • Iterative Development: I continue expanding and relabeling the dataset between model versions to improve performance under more difficult launch conditions.
